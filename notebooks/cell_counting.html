def voxel_to_um(voxel_dim, val, val_pos):
    if val_pos == 0:
        return val * voxel_dim[0]
    elif val_pos == 1:
        return val * voxel_dim[1]
    elif val_pos == 2:
        return val * voxel_dim[2]
    else:
        raise ValueError('val_pos must be from [0,1,2]')

                slice_min = voxel_to_um(voxel_dim, pred[0], 0)
